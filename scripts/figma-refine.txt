const text=(p,value,w,size=16,serif=false,color='ink')=>{const n=mark(figma.createText());n.fontName={family:serif?'DM Serif Display':'Manrope',style:'Regular'};n.characters=value;n.fontSize=size;n.lineHeight={unit:'PERCENT',value:serif?115:160};n.fills=[paint(color)];n.resize(w,10);n.textAutoResize='HEIGHT';p.appendChild(n);return n;};
const button=await figma.getNodeByIdAsync(STATE.components.button);
for(const mobile of [false,true]){

  const reviews=auto(home,'Travel stories',home.width,'VERTICAL',mobile?22:70);

   text(c,'★★★★★',c.width-48,16,false,'pink');text(c,'“'+quote+'”',c.width-48,15);text(c,name+' · Sample review',c.width-48,12);

 }

}
// Give repeated screen navigation a genuine reusable component per viewport.
const navComponents={};
for(const mobile of [false,true]){
 const name='M&M / Navigation / '+(mobile?'Mobile':'Desktop');let component=page.findOne(n=>n.type==='COMPONENT'&&n.name===name);
 if(!component){const source=(await figma.getNodeByIdAsync(STATE.wrappers['Home '+(mobile?'Mobile':'Desktop')])).children.find(n=>n.name==='Navigation');component=mark(figma.createComponent());component.name=name;component.resize(source.width,source.height);component.layoutMode='HORIZONTAL';component.primaryAxisSizingMode='FIXED';component.counterAxisSizingMode='AUTO';component.paddingTop=component.paddingBottom=mobile?20:24;component.paddingLeft=component.paddingRight=mobile?20:45;component.itemSpacing=source.itemSpacing;component.counterAxisAlignItems='CENTER';component.fills=[paint('paper')];for(const child of source.children){const c=child.clone();component.appendChild(c);made.push(c.id,...('findAll'in c?c.findAll(()=>true).map(x=>x.id):[]));}page.appendChild(component);component.x=1800;component.y=mobile?-1600:-1900;}
 navComponents[mobile?'mobile':'desktop']=component.id;
 for(const [name,id] of Object.entries(STATE.wrappers).filter(([name])=>name.endsWith(mobile?'Mobile':'Desktop'))){const screen=await figma.getNodeByIdAsync(id);const old=screen.children.find(n=>n.name==='Navigation'&&n.type==='FRAME');if(old){const index=screen.children.indexOf(old),instance=mark(component.createInstance());screen.insertChild(index,instance);changed.push(old.id,screen.id);old.remove();}}
